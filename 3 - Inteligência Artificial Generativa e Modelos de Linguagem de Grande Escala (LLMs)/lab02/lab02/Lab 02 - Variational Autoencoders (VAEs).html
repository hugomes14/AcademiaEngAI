Ex03: Função para definir a semente de reprodutibilidade

Para garantir resultados consistentes, definimos uma função que fixa as sementes dos vários geradores de números aleatórios:

import random

def set_seed(seed_value=42):
    """Define a semente para reprodutibilidade."""
    np.random.seed(seed_value)
    torch.manual_seed(seed_value)
    torch.cuda.manual_seed(seed_value)
    torch.cuda.manual_seed_all(seed_value)  # se estiveres a utilizar multi-GPU.
    random.seed(seed_value)
    os.environ['PYTHONHASHSEED'] = str(seed_value)

    # Comportamento determinístico na GPU (pode diminuir o desempenho)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False

set_seed()

Ex05: Definir o modelo VAE

O Variational Autoencoder (VAE) é composto por duas partes principais:

  • Encoder: Mapeia a entrada para um espaço latente de dimensão latent_dim (por meio de parâmetros de distribuição, média e log-variância).
  • Decoder: Reconstrói a entrada a partir do espaço latente.
class VAE(nn.Module):
    def __init__(self, latent_dim=2):
        super(VAE, self).__init__()
        # Encoder
        self.fc1 = nn.Linear(28 * 28, 400)
        self.fc21 = nn.Linear(400, latent_dim)  # Média (μ)
        self.fc22 = nn.Linear(400, latent_dim)  # Log-variância (logσ²)

        # Decoder
        self.fc3 = nn.Linear(latent_dim, 400)
        self.fc4 = nn.Linear(400, 28 * 28)

    def encode(self, x):
        h1 = F.relu(self.fc1(x))
        return self.fc21(h1), self.fc22(h1)

    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + eps * std

    def decode(self, z):
        h3 = F.relu(self.fc3(z))
        return torch.sigmoid(self.fc4(h3))

    def forward(self, x):
        mu, logvar = self.encode(x.view(-1, 28 * 28))
        z = self.reparameterize(mu, logvar)
        return self.decode(z), mu, logvar

Ex06: Definir a função de loss (Beta-VAE)

A loss function padrão de um VAE é constituída por duas partes:

  • Reconstruction Loss (BCE): Avalia a diferença entre a imagem original e a reconstruída.
  • KL Divergence (KLD): Força a distribuição latente a aproximar-se de uma distribuição normal padrão.

No caso de um Beta-VAE, adicionamos um fator () para controlar a ênfase no termo KL:

beta = 1.0

def loss_function(recon_x, x, mu, logvar):
    BCE = F.binary_cross_entropy(recon_x, x.view(-1, 28 * 28), reduction='sum')
    KLD = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
    return BCE + beta * KLD

Ex07: Treinar o VAE com espaço latente de dimensão 2

Definimos o nosso modelo com latent_dim=2 e treinamo-lo durante 100 épocas. Dependendo do teu hardware, poderás ajustar o número de épocas ou o batch size para reduzir o tempo de treino.

latent_dim = 2
model = VAE(latent_dim=latent_dim).to(device)
optimizer = optim.Adam(model.parameters(), lr=1e-3)
num_epochs = 100

for epoch in range(num_epochs):
    model.train()
    train_loss = 0
    for data, _ in trainloader:
        data = data.to(device)
        optimizer.zero_grad()
        recon_batch, mu, logvar = model(data)
        loss = loss_function(recon_batch, data, mu, logvar)
        loss.backward()
        train_loss += loss.item()
        optimizer.step()
    print(f'Epoch {epoch + 1}, Loss: {train_loss / len(trainloader.dataset)}')

Ex08: Visualizar a reconstrução no espaço latente de dimensão 2 (Latent Space Visualization)

Para analisar o espaço latente no conjunto de teste, projetamos cada imagem em duas dimensões (pois latent_dim=2) e colorimos cada ponto com base no seu rótulo (0 a 9).

model.eval()
all_z = []
all_labels = []
with torch.no_grad():
    for data, labels in testloader:
        data = data.to(device)
        mu, logvar = model.encode(data.view(-1, 28 * 28))
        z = model.reparameterize(mu, logvar)
        z = z.cpu().numpy()

        all_z.append(z)
        all_labels.append(labels)

    all_z = np.concatenate(all_z)
    all_labels = np.concatenate(all_labels)

    plt.figure(figsize=(10, 8))
    scatter = plt.scatter(all_z[:, 0], all_z[:, 1], c=all_labels, 
                          cmap='tab10', alpha=0.5, edgecolors='k', s=20)
    plt.colorbar(scatter, ticks=range(10))
    plt.xlabel('Dimensão Latente 1')
    plt.ylabel('Dimensão Latente 2')
    plt.title('Visualização do Espaço Latente')
    plt.grid(True)
    plt.savefig('resultado_espaco_latente_2D.png', bbox_inches='tight', pad_inches=0.1)    
    print("Imagem guardada como 'resultado_espaco_latente_2D.png'")
    plt.show()

## Ex09: Visualizar a reconstrução no espaço latente de dimensão 2 (Continuous Value Adjustment)

Agora, vamos observar como a mudança contínua nos valores latentes influencia a aparência dos dígitos reconstruídos.
Criamos uma malha (grid) de pontos no espaço latente e descodificamos cada um deles:

with torch.no_grad():
    grid_x = np.linspace(-3, 3, 20)
    grid_y = np.linspace(-3, 3, 20)
    figure = np.zeros((28 * 20, 28 * 20))

    for i, yi in enumerate(grid_x):
        for j, xi in enumerate(grid_y):
            z_sample = torch.Tensor([[xi, yi]]).to(device)
            x_decoded = model.decode(z_sample)
            digit = x_decoded.view(28, 28).cpu().numpy()
            figure[i * 28: (i + 1) * 28, j * 28: (j + 1) * 28] = digit

    plt.figure(figsize=(15, 15))
    plt.imshow(figure, cmap='gray')
    plt.title('Visualização de Ajuste Contínuo de Valores')
    plt.savefig('resultado_ajuste_continuo_valores_2D.png', bbox_inches='tight', pad_inches=0.1)    
    print("Imagem guardada como 'resultado_ajuste_continuo_valores_2D.png'")
    plt.show()

Cada linha e coluna representam valores no espaço latente, e cada célula corresponde ao dígito gerado para o ponto ((x_i, y_i)) na malha.

Ex10: Treinar e Visualizar a reconstrução com espaço latente de dimensão 128

Podemos repetir o processo de treino e reconstrução, mas agora com um espaço latente maior, por exemplo latent_dim=128.
O VAE pode, assim, capturar mais variação nos dados, mas a sua representação em 2D torna-se mais complexa.

latent_dim = 128
model = VAE(latent_dim=latent_dim).to(device)
optimizer = optim.Adam(model.parameters(), lr=1e-3)
num_epochs = 100

for epoch in range(num_epochs):
    model.train()
    train_loss = 0
    for data, _ in trainloader:
        data = data.to(device)
        optimizer.zero_grad()
        recon_batch, mu, logvar = model(data)
        loss = loss_function(recon_batch, data, mu, logvar)
        loss.backward()
        train_loss += loss.item()
        optimizer.step()
    print(f'Epoch {epoch + 1}, Loss: {train_loss / len(trainloader.dataset)}')

Para avaliar a qualidade da reconstrução:

model.eval()
with torch.no_grad():
    data, _ = next(iter(testloader))
    data = data.to(device)
    recon, _, _ = model(data)
    recon = recon.view(-1, 1, 28, 28).cpu().numpy()

    fig, axes = plt.subplots(2, 8, figsize=(15, 4))
    for i in range(8):
        axes[0, i].imshow(data[i].cpu().numpy().squeeze(), cmap='gray')
        axes[0, i].set_title("Original")
        axes[0, i].axis('off')

        axes[1, i].imshow(recon[i].squeeze(), cmap='gray')
        axes[1, i].set_title("Reconstruído")
        axes[1, i].axis('off')

    plt.savefig('reconstrucao_128d.png', bbox_inches='tight', pad_inches=0.1)
    print("Imagem final guardada como 'reconstrucao_128d.png'")
    plt.show()
